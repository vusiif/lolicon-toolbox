find_package(Qt6 REQUIRED COMPONENTS Concurrent)
add_library(CharacterArtLogic STATIC CharacterArt.h CharacterArt.cpp)
target_include_directories(CharacterArtLogic PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(CharacterArtLogic PUBLIC Qt6::Gui)
set(maple_font "${PROJECT_SOURCE_DIR}/third_party/MapleMonoNL-NF-CN-unhinted/MapleMonoNL-NFMono-CN-Regular.ttf")
file(SHA256 "${maple_font}" maple_hash)
if(NOT maple_hash STREQUAL "f01646c491179bc0cc547d9d5e6e6dc3e3894bd0c13ac2bffaf6ae4acb04b0a1")
    message(FATAL_ERROR "Maple font changed: verify version, license and density calibration before updating the pinned SHA256")
endif()
qt_add_resources(CharacterArtLogic maple PREFIX "/characterart" BASE "${PROJECT_SOURCE_DIR}/third_party/MapleMonoNL-NF-CN-unhinted" FILES "${maple_font}")
add_library(ToolCharacterArt STATIC CharacterArtPage.h CharacterArtPage.cpp)
target_include_directories(ToolCharacterArt PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
